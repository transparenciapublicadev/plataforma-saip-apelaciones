  <TabItem label="Servicios">
    ```typescript
    import { TestBed } from '@angular/core/testing';
    import { HttpTestingController, provideHttpClientTesting }
      from '@angular/common/http/testing';

    describe('SolicitudService', () => {
      let service: SolicitudService;
      let httpMock: HttpTestingController;

      beforeEach(() => {
        TestBed.configureTestingModule({
          providers: [
            provideHttpClient(),
            provideHttpClientTesting(),
          ],
        });
        service = TestBed.inject(SolicitudService);
        httpMock = TestBed.inject(HttpTestingController);
      });

      afterEach(() => httpMock.verify());

      it('debería obtener solicitudes', () => {
        service.findAll().subscribe(solicitudes => {
          expect(solicitudes).toHaveLength(2);
        });

        const req = httpMock.expectOne('/api/solicitudes');
        expect(req.request.method).toBe('GET');
        req.flush([mockSolicitud1, mockSolicitud2]);
      });
    });
    ```
  </TabItem>
  <TabItem label="Guards">
    ```typescript
    describe('authGuard', () => {
      it('debería permitir acceso si hay sesión', () => {
        const authService = { estaLogueado: () => true };
        TestBed.configureTestingModule({
          providers: [
            { provide: AuthService, useValue: authService },
          ],
        });

        const result = TestBed.runInInjectionContext(() =>
          authGuard({} as any, {} as any));

        expect(result).toBe(true);
      });

      it('debería redirigir a login sin sesión', () => {
        const authService = { estaLogueado: () => false };
        const router = { navigate: vi.fn() };

        TestBed.configureTestingModule({
          providers: [
            { provide: AuthService, useValue: authService },
            { provide: Router, useValue: router },
          ],
        });

        const result = TestBed.runInInjectionContext(() =>
          authGuard({} as any, {} as any));

        expect(result).toBe(false);
        expect(router.navigate).toHaveBeenCalledWith(['/login']);
      });
    });
    ```

## Ejecutar Tests

```bash
# Todos los tests
cd frontend && npm test

# Tests en modo watch
npm run test:watch

# Tests con cobertura
npm run test:coverage
```

## Convenciones

| Regla | Valor |
|-------|-------|
| Archivo de test | `*.spec.ts` junto al archivo fuente |
| Nomenclatura | `debería + verbo + resultado esperado` |
| Mocks | Servicios y dependencias siempre mockeados |
| Cobertura objetivo | 80%+ |
| Framework | Vitest + Angular Testing Library |
